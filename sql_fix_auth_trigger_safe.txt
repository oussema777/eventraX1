-- FIX: Robust Auth Trigger Function
-- This script implements a "safe" version of the handle_new_user trigger function.
-- It uses COALESCE to prevent NULL errors and ON CONFLICT to prevent crashes.

CREATE OR REPLACE FUNCTION public.handle_new_user() 
RETURNS trigger 
LANGUAGE plpgsql 
SECURITY DEFINER SET search_path = auth, public
AS $$
BEGIN
  -- Wrap in an exception block to ensure the login never fails
  -- even if profile creation has an issue.
  BEGIN
    INSERT INTO public.profiles (id, email, full_name, avatar_url, role)
    VALUES (
      new.id,
      new.email,
      COALESCE(new.raw_user_meta_data->>'full_name', new.raw_user_meta_data->>'name', 'New User'),
      COALESCE(new.raw_user_meta_data->>'avatar_url', ''),
      'user'
    )
    ON CONFLICT (id) DO UPDATE
    SET email = EXCLUDED.email,
        full_name = COALESCE(EXCLUDED.full_name, public.profiles.full_name),
        avatar_url = COALESCE(EXCLUDED.avatar_url, public.profiles.avatar_url),
        updated_at = NOW();
  EXCEPTION WHEN OTHERS THEN
    RETURN NEW;
  END;

  RETURN NEW;
END;
$$;
