-- EVENTRA NOTIFICATION CENTER SCHEMA
-- Safe to re-run: uses IF NOT EXISTS and DROP IF EXISTS on policies.

-- 1. Create Notification Settings Table
CREATE TABLE IF NOT EXISTS public.event_notification_settings (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_id UUID REFERENCES public.events(id) ON DELETE CASCADE,
    trigger_type TEXT NOT NULL,
    is_email_enabled BOOLEAN DEFAULT true,
    is_bell_enabled BOOLEAN DEFAULT true,
    custom_subject TEXT,
    custom_body TEXT,
    form_id UUID REFERENCES public.event_forms(id) ON DELETE SET NULL,
    delay_hours INTEGER DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(event_id, trigger_type)
);

-- 2. Create Notification Logs for Audit/Tracking
CREATE TABLE IF NOT EXISTS public.notification_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_id UUID REFERENCES public.events(id) ON DELETE CASCADE,
    recipient_id UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
    trigger_type TEXT NOT NULL,
    channel TEXT CHECK (channel IN ('email', 'bell')),
    status TEXT DEFAULT 'sent',
    sent_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Create Form Invitations Table
CREATE TABLE IF NOT EXISTS public.event_form_invitations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    form_id UUID REFERENCES public.event_forms(id) ON DELETE CASCADE,
    attendee_id UUID REFERENCES public.event_attendees(id) ON DELETE CASCADE,
    status TEXT DEFAULT 'sent',
    last_reminded_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(form_id, attendee_id)
);

-- 4. Enable RLS (safe to run even if already enabled)
ALTER TABLE public.event_notification_settings ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.notification_logs ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.event_form_invitations ENABLE ROW LEVEL SECURITY;

-- 5. Drop existing policies first to avoid duplicate errors, then recreate
DROP POLICY IF EXISTS "Event owners can manage notification settings" ON public.event_notification_settings;
CREATE POLICY "Event owners can manage notification settings"
ON public.event_notification_settings
FOR ALL
TO authenticated
USING (
    EXISTS (
        SELECT 1 FROM public.events
        WHERE events.id = event_notification_settings.event_id
        AND (events.owner_id = auth.uid() OR events.organizer_id = auth.uid())
    )
);

DROP POLICY IF EXISTS "Event owners can view notification logs" ON public.notification_logs;
CREATE POLICY "Event owners can view notification logs"
ON public.notification_logs
FOR SELECT
TO authenticated
USING (
    EXISTS (
        SELECT 1 FROM public.events
        WHERE events.id = notification_logs.event_id
        AND (events.owner_id = auth.uid() OR events.organizer_id = auth.uid())
    )
);

-- Allow authenticated users to insert their own notification logs
DROP POLICY IF EXISTS "Authenticated users can insert notification logs" ON public.notification_logs;
CREATE POLICY "Authenticated users can insert notification logs"
ON public.notification_logs
FOR INSERT
TO authenticated
WITH CHECK (true);

-- 6. Trigger to update updated_at
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ language 'plpgsql';

DROP TRIGGER IF EXISTS update_event_notification_settings_updated_at ON public.event_notification_settings;
CREATE TRIGGER update_event_notification_settings_updated_at
    BEFORE UPDATE ON public.event_notification_settings
    FOR EACH ROW
    EXECUTE PROCEDURE update_updated_at_column();
