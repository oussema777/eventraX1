-- 1. Create the table
CREATE TABLE IF NOT EXISTS public.event_form_submissions (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    created_at TIMESTAMPTZ DEFAULT now() NOT NULL,
    event_id UUID NOT NULL REFERENCES public.events(id) ON DELETE CASCADE,
    form_id UUID NOT NULL REFERENCES public.event_forms(id) ON DELETE CASCADE,
    attendee_id UUID REFERENCES public.event_attendees(id) ON DELETE SET NULL,
    data JSONB DEFAULT '{}'::jsonb,
    metadata JSONB DEFAULT '{}'::jsonb
);

-- 2. Enable Row Level Security (RLS)
ALTER TABLE public.event_form_submissions ENABLE ROW LEVEL SECURITY;

-- 3. Create Policies

-- Allow anyone to submit a form (Public Access)
CREATE POLICY "Anyone can submit forms" 
    ON public.event_form_submissions FOR INSERT 
    WITH CHECK (true);

-- Allow Event Owners to view submissions for their events
CREATE POLICY "Owners can view their form submissions" 
    ON public.event_form_submissions FOR SELECT 
    USING (
        EXISTS (
            SELECT 1 FROM public.events 
            WHERE events.id = event_form_submissions.event_id 
            AND events.owner_id = auth.uid()
        )
    );

-- Allow Attendees to view their own submissions (Optional)
CREATE POLICY "Attendees can view own submissions" 
    ON public.event_form_submissions FOR SELECT 
    USING (
        attendee_id IS NOT NULL AND 
        attendee_id IN (
            SELECT id FROM public.event_attendees 
            WHERE profile_id = auth.uid()
        )
    );
