-- FINAL FIX: RLS for Notification Settings (Simplified)
-- Based on the real schema where only 'owner_id' exists in public.events.

-- 1. Remove all old policies
DROP POLICY IF EXISTS "Event owners can manage notification settings" ON public.event_notification_settings;
DROP POLICY IF EXISTS "Users can view notification settings for their events" ON public.event_notification_settings;
DROP POLICY IF EXISTS "Users can manage notification settings for their events" ON public.event_notification_settings;
DROP POLICY IF EXISTS "Manage own notification settings" ON public.event_notification_settings;
DROP POLICY IF EXISTS "notification_settings_owner_policy" ON public.event_notification_settings;

-- 2. Create the most direct policy possible
-- This allows anyone to READ (for the engine to work) but only OWNERS to WRITE
CREATE POLICY "notification_settings_read_policy"
ON public.event_notification_settings
FOR SELECT
TO authenticated
USING (true);

CREATE POLICY "notification_settings_write_policy"
ON public.event_notification_settings
FOR ALL
TO authenticated
USING (
  event_id IN (
    SELECT id FROM public.events WHERE owner_id = auth.uid()
  )
)
WITH CHECK (
  event_id IN (
    SELECT id FROM public.events WHERE owner_id = auth.uid()
  )
);

-- 3. Ensure RLS is active
ALTER TABLE public.event_notification_settings ENABLE ROW LEVEL SECURITY;

-- 4. Grant full permissions
GRANT ALL ON public.event_notification_settings TO authenticated;
GRANT ALL ON public.event_notification_settings TO service_role;
